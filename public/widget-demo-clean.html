    // SETTINGS PERSISTENCE
    const STORAGE_KEY = 'emojiRain_settings';

    function saveSettings() {
      const settings = {
        emojis: document.getElementById('emojiInput').value,
        containerZIndex: document.getElementById('containerZIndex').value,
        modalZIndex: document.getElementById('modalZIndex').value,
        baseSpeed: document.getElementById('baseSpeed').value,
        parallax: document.getElementById('parallaxCheckbox').checked,
        blur: document.getElementById('blurCheckbox').checked,
        spawnRate: document.getElementById('spawnRate').value,
        maxEmojis: document.getElementById('maxEmojis').value,
        frontRatio: document.getElementById('frontRatio').value,
        middleRatio: document.getElementById('middleRatio').value,
        backRatio: document.getElementById('backRatio').value
      };
      
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
      } catch (e) {
        console.warn('Failed to save settings:', e);
      }
    }

    function loadSettings() {
      try {
        const saved = localStorage.getItem(STORAGE_KEY);
        if (!saved) return;
        
        const settings = JSON.parse(saved);
        if (settings.emojis) document.getElementById('emojiInput').value = settings.emojis;
        if (settings.containerZIndex) document.getElementById('containerZIndex').value = settings.containerZIndex;
        if (settings.modalZIndex) document.getElementById('modalZIndex').value = settings.modalZIndex;
        if (settings.baseSpeed) document.getElementById('baseSpeed').value = settings.baseSpeed;
        if (settings.parallax !== undefined) document.getElementById('parallaxCheckbox').checked = settings.parallax;
        if (settings.blur !== undefined) document.getElementById('blurCheckbox').checked = settings.blur;
        if (settings.spawnRate) document.getElementById('spawnRate').value = settings.spawnRate;
        if (settings.maxEmojis) document.getElementById('maxEmojis').value = settings.maxEmojis;
        if (settings.frontRatio) document.getElementById('frontRatio').value = settings.frontRatio;
        if (settings.middleRatio) document.getElementById('middleRatio').value = settings.middleRatio;
        if (settings.backRatio) document.getElementById('backRatio').value = settings.backRatio;
        
        // Update config
        parallaxEnabled = settings.parallax !== false;
        blurEnabled = settings.blur === true;
        parallaxConfig.baseSpeed = parseInt(settings.baseSpeed) || 50;
        parallaxConfig.layerRatios.front = parseInt(settings.frontRatio) || 100;
        parallaxConfig.layerRatios.middle = parseInt(settings.middleRatio) || 300;
        parallaxConfig.layerRatios.back = parseInt(settings.backRatio) || 70;
      } catch (e) {
        console.warn('Failed to load settings:', e);
      }
    }

    // EVENT LISTENERS
    document.getElementById('parallaxCheckbox').addEventListener('change', (e) => {
      parallaxEnabled = e.target.checked;
      saveSettings();
    });

    document.getElementById('blurCheckbox').addEventListener('change', (e) => {
      blurEnabled = e.target.checked;
      saveSettings();
    });

    document.getElementById('baseSpeed').addEventListener('input', (e) => {
      parallaxConfig.baseSpeed = parseInt(e.target.value) || 50;
      saveSettings();
    });

    document.getElementById('continuousMode').addEventListener('change', (e) => {
      if (e.target.checked) {
        startContinuous();
      } else {
        stopContinuous();
      }
    });

    document.getElementById('frontRatio').addEventListener('input', (e) => {
      parallaxConfig.layerRatios.front = parseInt(e.target.value) || 100;
      saveSettings();
    });

    document.getElementById('middleRatio').addEventListener('input', (e) => {
      parallaxConfig.layerRatios.middle = parseInt(e.target.value) || 300;
      saveSettings();
    });

    document.getElementById('backRatio').addEventListener('input', (e) => {
      parallaxConfig.layerRatios.back = parseInt(e.target.value) || 70;
      saveSettings();
    });

    document.getElementById('emojiInput').addEventListener('input', saveSettings);
    document.getElementById('containerZIndex').addEventListener('input', saveSettings);
    document.getElementById('modalZIndex').addEventListener('input', saveSettings);
    document.getElementById('spawnRate').addEventListener('input', saveSettings);
    document.getElementById('maxEmojis').addEventListener('input', saveSettings);

    // INIT
    window.addEventListener('DOMContentLoaded', () => {
      loadSettings();
